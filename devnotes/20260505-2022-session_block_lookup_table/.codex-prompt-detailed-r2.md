### microbenchmark 分離 + 相対比較

```python
# 本線 unit test 例 (相対比較)
def test_compute_bucket_for_bar_at_least_15pct_faster_than_oracle():
    """同一プロセス内で lookup 版が現行 oracle 比で 15%+ 高速。 環境差吸収のため相対比較."""
    n = 100_000
    bar_times = [datetime(2026, 1, 1, h % 24, tzinfo=timezone.utc) for h in range(n)]

    # oracle (test 内 self-contained dict iter)
    def oracle(bt):
        h = bt.hour
        for bucket, (start, end) in BLOCK_BUCKET_RANGES_UTC.items():
            if start <= h < end:
                return bucket
        raise RuntimeError(...)

    # timing
    t0 = time.perf_counter_ns(); [oracle(bt) for bt in bar_times]; t_oracle = time.perf_counter_ns() - t0
    t0 = time.perf_counter_ns(); [compute_bucket_for_bar(bt) for bt in bar_times]; t_lookup = time.perf_counter_ns() - t0

    # lookup median per-call が oracle 比 15%+ 高速 = ratio < 0.85
    ratio = t_lookup / t_oracle
    assert ratio < 0.85, f"lookup ratio={ratio:.3f} (oracle 比 {(1-ratio)*100:.1f}% 高速、 期待: 15%+)"
```
