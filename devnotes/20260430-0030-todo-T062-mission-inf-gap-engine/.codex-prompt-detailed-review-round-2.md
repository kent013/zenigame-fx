#### 1. sentinel +inf 撤廃 + constraint_violation 新設 (Round 1 [C1] / [S1])

```python
# MISSION_GAP_INFEASIBLE_SENTINEL は撤廃 (削除)
MISSION_SIGNED_MARGIN_INFEASIBLE_SENTINEL: Final[float] = float("-inf")  # 維持 (archive eviction 用)

@dataclass(frozen=True)
class MissionGapResult:
    mission_inf_gap: float          # 純粋 § 6.4 式、 [0, +inf)、 sentinel なし
    constraint_violation: float     # 新設、 [0, +inf)、 feasible=0.0、 infeasible=violation 量
    mission_margin: float           # DEPRECATED-COMPAT: synthesis § 8.3 命名通り
    mission_signed_margin: float    # 実運用 SSOT、 archive CA #5 ordering 用、 -inf=infeasible
    per_metric_shortfall: Mapping[str, float]  # MappingProxyType wrap、 raw 計算値
    is_feasible: bool

def compute_constraint_violation(slacks, is_feasible) -> float:
    """Deb 2000 用 violation 量 (有限値、 sentinel なし)."""
    _validate_slacks_dict(slacks)
    if is_feasible:
        return 0.0
    return max(max(0.0, -slacks[k]) for k in MISSION_INF_GAP_METRIC_KEYS)
```
