class StagePassStatus(StrEnum):
    PASS = "pass"
    FAIL = "fail"
    PENDING = "pending"


def evaluate_stage_c(individual_input, *, evaluate_fn, live_criteria, spread_stress_supported=False):
    if spread_stress_supported:
        stress_cf_result = evaluate_fn(apply_spread_stress(...))
        stress_pass = StagePassStatus.PASS if stress_cf_result.gate_pass else StagePassStatus.FAIL
    else:
        stress_cf_result = None
        stress_pass = StagePassStatus.PENDING
    
    if live_criteria_pass and stress_pass == PASS and cross_pair_pass == PASS:
        mission_pass = PASS
    elif stress_pass == PENDING:
        mission_pass = PENDING
    else:
        mission_pass = FAIL
