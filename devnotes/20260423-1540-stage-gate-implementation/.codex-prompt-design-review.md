```python
        if self.wf_embargo_days < 0:
            raise ValueError("wf_embargo_days must be >= 0")
        if self.spread_stress_multiplier <= 1.0:
            raise ValueError(f"spread_stress_multiplier must be > 1.0: {self.spread_stress_multiplier}")
        # frozen dataclass でも __post_init__ で defensive copy
        object.__setattr__(self, "live_criteria", dict(self.live_criteria))
```

### 2.4 `make_wf_folds`

```python
def make_wf_folds(
    bars: list[PriceBar],
    train_days: int,
    test_days: int,
    step_days: int,
    embargo_days: int,
) -> list[tuple[list[PriceBar], list[PriceBar]]]:
    """observed-day index ベースで (train_bars, test_bars) のリストを返す.

    Args:
        bars: 評価対象 bars (bar_time 昇順前提、関数内 assert)。
        train_days: train 区間の観測日数 (>= 1)。
        test_days:  test 区間の観測日数 (>= 1)。
        step_days:  fold 間の観測日数 step (>= 1)。
        embargo_days: train と test の間の隔離観測日数 (>= 0)。

    Returns:
        各 fold の (train_bars, test_bars) tuple のリスト。
        train_days + embargo_days + test_days > n_unique_dates のとき空 list。

    Raises:
        ValueError: 引数が範囲外、bars が空、bars が時系列昇順でない。
    """
```

**算法**:
```python
def make_wf_folds(bars, train_days, test_days, step_days, embargo_days):
    if train_days < 1 or test_days < 1 or step_days < 1:
        raise ValueError(...)
    if embargo_days < 0:
        raise ValueError(...)
    if not bars:
        return []
    # 時系列昇順 assert
    for i in range(1, len(bars)):
        if bars[i].bar_time < bars[i-1].bar_time:
            raise ValueError("bars must be ascending by bar_time")

    # ユニーク UTC date 列
    seen: set[date] = set()
    sorted_dates: list[date] = []
    for b in bars:
        d = b.bar_time.date()
        if d not in seen:
            seen.add(d)
            sorted_dates.append(d)

    n_days = len(sorted_dates)
    fold_len = train_days + embargo_days + test_days
    if fold_len > n_days:
        return []

    # 各 date → bars インデックス範囲を一度だけ計算（O(n) 最適化）
    bars_by_date: dict[date, list[PriceBar]] = {}
    for b in bars:
        bars_by_date.setdefault(b.bar_time.date(), []).append(b)

    folds: list[tuple[list[PriceBar], list[PriceBar]]] = []
    k = 0
    while True:
        train_start = step_days * k
        train_end = train_start + train_days  # exclusive
        test_start = train_end + embargo_days
        test_end = test_start + test_days     # exclusive
        if test_end > n_days:
            break
        train_dates = sorted_dates[train_start:train_end]
        test_dates  = sorted_dates[test_start:test_end]
        train_bars: list[PriceBar] = []
        for d in train_dates:
            train_bars.extend(bars_by_date[d])
        test_bars: list[PriceBar] = []
        for d in test_dates:
            test_bars.extend(bars_by_date[d])
        folds.append((train_bars, test_bars))
        k += 1
    return folds
```

**embargo bars はどちらにも含めない**ことが §3.6 の決定。アルゴリズムは
embargo dates をスキップして test_dates を取るので自然に成立。

```python
if not lc_pass["sharpe"]:

if not lc_pass["trade_count_min"]:

        res = run_backtest(bars_holdout, strategy, broker, stress_config)
        bt = compute_metrics(res.trades, res.equity_curve)
        s_sharpe = float(bt.sharpe) if bt.sharpe is not None else None
        stress_payload["sharpe"] = s_sharpe
        stress_payload["total_pnl"] = float(bt.total_pnl)
        stress_payload["max_drawdown_frac"] = float(bt.max_drawdown_pct) / 100.0
        stress_payload["trade_count"] = bt.trade_count
        if base_sharpe is not None and s_sharpe is not None:
            stress_payload["sharpe_degradation"] = base_sharpe - s_sharpe
        stress_payload["pnl_degradation"] = base_total_pnl - float(bt.total_pnl)
        # hard gate (AND)
        if bt.trade_count < int(lc["trade_count_min"]):
            reasons.append("spread_stress.trade_count<min")
        if float(bt.total_pnl) < float(stage_config.spread_stress_min_total_pnl):
            reasons.append("spread_stress.total_pnl<0")
        if s_sharpe is None or s_sharpe < float(stage_config.spread_stress_min_sharpe):
            reasons.append("spread_stress.sharpe<0")
    except Exception as exc:
        logger.warning("stage_c.stress_failure", genome=genome.name, error=str(exc))
        stress_payload["skipped"] = True
        reasons.append("spread_stress_skipped")

# 5) cross-pair shadow hook
cross_pair_payload: dict[str, object] = {"skipped": True, "result": None}
if cross_pair_evaluator is not None:
    if cross_pair_inputs is None:
        raise ValueError("cross_pair_inputs required when cross_pair_evaluator is set")
    cp_result = cross_pair_evaluator.evaluate(
        genome=genome,
        target_pair=cross_pair_inputs["target_pair"],
        pair_bars_map=cross_pair_inputs["pair_bars_map"],
        meta_map=cross_pair_inputs["meta_map"],
        backtest_config=backtest_config,
    )
    cross_pair_payload["skipped"] = False
    cross_pair_payload["result"] = cp_result
    # Phase 2 では shadow only。passed には影響させない

passed = len(reasons) == 0
elapsed = _time.perf_counter() - start

metrics_envelope = {
    "stage": "C",
    "genome_name": genome.name,
    "n_bars": len(bars_holdout),
    "wall_time_seconds": elapsed,
    "payload": {
        "sharpe": base_sharpe,
        "total_pnl": base_total_pnl,
        "max_drawdown_frac": base_max_dd_frac,
        "trade_count": base_trade_count,
        "live_criteria_pass": lc_pass,
        "intraday_compliant": intraday_compliant,
        "overnight_violations": overnight_violations,
        "stress": stress_payload,
        "cross_pair": cross_pair_payload,
    },
}
return StageResult(stage="C", passed=passed, metrics=metrics_envelope,
                   reason_codes=tuple(reasons))
```
