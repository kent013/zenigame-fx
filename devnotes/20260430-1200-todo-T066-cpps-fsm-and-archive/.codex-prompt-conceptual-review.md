```

1. new_state = update_push_pull_state(
       prev_state,
       generation_no=g,
       feasible_ratio_ema=r_feas,
```

### 5.2 遷移関数

```python
def update_push_pull_state(
    prev: PushPullState,
    *,
    generation_no: int,
    feasible_ratio_ema: float,
    force_g: int,                    # gen=64 で 44、 gen=48 で再校正
    theta_switch: float,             # smoke 後再校正、 仮 0.40
    consecutive_n: int = 3,          # zenigame 踏襲
) -> PushPullState:
    """Push → Pull 一方向遷移を計算.

    一方向性 (push → pull のみ): 一度 pull に入ったら pull のまま
    強制遷移: generation_no >= force_g なら pull 強制 (forced_switch=True)
    通常遷移: feasible_ratio_ema >= theta_switch かつ
              consecutive_count >= consecutive_n の場合
    """
    if prev.phase == "pull":
        # 一方向、 pull 維持
        return replace(prev, last_feasible_ratio_ema=feasible_ratio_ema)

    # push → pull 判定
    if generation_no >= force_g:
        return PushPullState(
            phase="pull",
            consecutive_count=0,
            switch_generation=generation_no,
            last_feasible_ratio_ema=feasible_ratio_ema,
            forced_switch=True,
        )

    if feasible_ratio_ema >= theta_switch:
        new_count = prev.consecutive_count + 1
        if new_count >= consecutive_n:
            return PushPullState(
                phase="pull",
                consecutive_count=new_count,
                switch_generation=generation_no,
                last_feasible_ratio_ema=feasible_ratio_ema,
                forced_switch=False,
            )
        return replace(prev, consecutive_count=new_count, last_feasible_ratio_ema=feasible_ratio_ema)
    else:
        # 連続失敗、 count リセット
        return replace(prev, consecutive_count=0, last_feasible_ratio_ema=feasible_ratio_ema)
```

### 5.3 CA/DA capacity 計算

```python
PUSH_CA_RATIO = 84 / 192   # = 0.4375
PULL_CA_RATIO = 120 / 192  # = 0.625

def compute_ca_da_capacities(pop_size: int, phase: Literal["push", "pull"]) -> tuple[int, int]:
    """state 依存 CA/DA capacity (端数規約: CA=round(pop×ratio), DA=pop-CA)."""
    if pop_size < 2:
        raise ValueError(f"pop_size must be >= 2, got {pop_size}")
    ratio = PUSH_CA_RATIO if phase == "push" else PULL_CA_RATIO
    ca = round(pop_size * ratio)
    da = pop_size - ca
    return ca, da
```

合計 = pop_size 一致を保証 (端数規約 synthesis § 7.3)。

## 6. archive_role 判定 (3 層流入 + ineligible)

```python
def determine_archive_role(
    bc_result: BCEvaluationResult | None,
) -> Literal["mission_pass", "progress_pass", "score_bypass_candidate", "ineligible"]:
    """T064 出力から archive_role を一意に決定.

    階層: mission_pass > progress_pass > score_bypass_candidate > ineligible
    - mission_pass: bc_result.mission_pass == StagePassStatus.PASS
    - progress_pass: bc_result.progress_pass == StagePassStatus.PASS (mission_pass=False の時)
    - score_bypass_candidate: pareto_axis_usable=True かつ b_pooled_cf is not None
      (= Stage B 評価済 + invariant_feasible)、 上記 2 層に該当しない場合
    - ineligible: bc_result is None (a_fail or B-invariant-fail)、 archive 流入対象外

    contract: caller (archive_admit) は最終的に品質床
    (margin_inf percentile <= 70) と pattern_max_share / per_run_max を適用してから admit.
    """
    if bc_result is None:
        return "ineligible"
    if bc_result.mission_pass == StagePassStatus.PASS:
        return "mission_pass"
    if bc_result.progress_pass == StagePassStatus.PASS:
        return "progress_pass"
    if bc_result.pareto_axis_usable and bc_result.b_pooled_cf is not None:
        return "score_bypass_candidate"
    return "ineligible"
```

### 6.1 Inflow targets

```python
@dataclass(frozen=True)
class InflowTargets:
    target_inflow: int           # 0.04 × pop
    per_run_max: int             # 0.06 × pop
    bypass_k: int                # clamp(target - n_mission - n_progress, [2, 6] or [4, 8])
    mode: Literal["normal", "emergency"]


def compute_inflow_targets(
    pop_size: int,
    n_mission: int,
    n_progress: int,
    *,
    mode: Literal["normal", "emergency"],
) -> InflowTargets:
    """synthesis § 8.2 確定式."""
    if pop_size == 192:
        target = 8
        per_run_max = 12
    elif pop_size == 256:
        target = 10
        per_run_max = 16
    else:
        raise ValueError(f"pop_size must be 192 or 256, got {pop_size}")

    if mode == "emergency":
        target += 2     # synthesis § 8.2 emergency: clamp(target+2 - n_m - n_p, 4, 8)
        bypass_min, bypass_max = 4, 8
    else:
        bypass_min, bypass_max = 2, 6

    bypass_k = max(bypass_min, min(bypass_max, target - n_mission - n_progress))
    return InflowTargets(
        target_inflow=target,
        per_run_max=per_run_max,
        bypass_k=bypass_k,
        mode=mode,
    )
```

## 7. Survivor → CA/DA partition

```python
def partition_survivors_to_ca_da(
    survivor_indices: Sequence[int],
    *,
    phase: Literal["push", "pull"],
    pop_size: int,
    sort_keys: Mapping[int, tuple],   # T065 sort_keys: (front_no, -crowding, genome_hash, index)
) -> tuple[tuple[int, ...], tuple[int, ...]]:
    """state 依存比率で survivor を CA / DA に分割.

    - CA = sort_keys 上位 ca_cap 体 (rank/crowding で優位な個体、 収束圧)
    - DA = 残り da_cap 体 (多様性保持)
    survivor_indices は T065 で既に sort_keys 順にソート済の前提
    (= run_generation_selection の sorted_eligible[:pop_size])

    入口契約: len(survivor_indices) <= pop_size、 上記順序保証
    """
    ca_cap, da_cap = compute_ca_da_capacities(pop_size, phase)
    survivors_list = list(survivor_indices)
    if len(survivors_list) > pop_size:
        raise ValueError(
            f"survivor count {len(survivors_list)} exceeds pop_size {pop_size}"
        )

    ca_indices = tuple(survivors_list[:ca_cap])
    da_indices = tuple(survivors_list[ca_cap:ca_cap + da_cap])
    # eligible が pop_size 未満の場合、 ca_indices + da_indices < pop_size (caller 認識)
    return ca_indices, da_indices
```

注: T065 GenerationSelectionResult.survivor_indices は既に sort_keys 順 (rank → -crowding → genome_hash → index) で並んでいる。 T066 はその順序を尊重して上位 ca_cap を CA に、 残りを DA に振り分ける。

### 8.2 archive_admit 実装

```python
def archive_admit(
    archive_state: ArchiveState,
    candidates: Mapping[int, ArchiveMember],
    *,
    pop_size: int,
    mode: Literal["normal", "emergency"],
) -> tuple[ArchiveState, AdmissionReport]:
    """3 層流入 + ハード制約 + 品質床.

    Phase 1: pure function、 archive_state は immutable に置換
    Phase 2: T067 で run_ga.py 配線、 emergency mode 判定 caller 責務
    """
    # 1. 階層別カウント
    n_mission = sum(1 for m in candidates.values() if m.archive_role == "mission_pass")
    n_progress = sum(1 for m in candidates.values() if m.archive_role == "progress_pass")
    targets = compute_inflow_targets(pop_size, n_mission, n_progress, mode=mode)

    # 2. mission_pass 全件 (per_run_max まで)
    admitted = []
    mission_admits = sorted(
        (m for m in candidates.values() if m.archive_role == "mission_pass"),
        key=lambda m: m.mission_signed_margin, reverse=True,    # 大が上位
    )
    admitted.extend(mission_admits)

    # 3. progress_pass: worst_gap 昇順で remaining inflow まで
    remaining = max(0, targets.target_inflow - len(admitted))
    progress_admits = sorted(
        (m for m in candidates.values() if m.archive_role == "progress_pass"),
        key=lambda m: m.gate_worst_gap,                          # 小が上位
    )[:remaining]
    admitted.extend(progress_admits)

    # 4. score_bypass: bypass_k 体、 品質床通過のみ
    bypass_pool = sorted(
        (
            m for m in candidates.values()
            if m.archive_role == "score_bypass_candidate"
            and m.invariant_feasible
            and m.margin_inf <= MARGIN_INF_PERCENTILE_70_THRESHOLD  # caller 計算済
        ),
        key=lambda m: m.gate_worst_gap,
    )[:targets.bypass_k]
    # archive_role を score_bypass に確定 (admission 時点で score_bypass_candidate → score_bypass)
    bypass_admits = [replace(m, archive_role="score_bypass", not_score_bypass=False) for m in bypass_pool]
    admitted.extend(bypass_admits)

    # 5. ハード制約: per_run_max / pattern_max_share / recency_floor
    admitted_filtered = apply_hard_constraints(admitted, archive_state, pop_size)

    # 6. CA / DA 振り分け (mission_pass / progress_pass は CA、 score_bypass は CA、 DA 振り分けは別途 — Phase 1 では archive_target を呼出側で確定)
    new_state = _merge_admits_into_archive(archive_state, admitted_filtered)

    # 7. eviction (CA / DA capacity 超過時)
    archive_total, ca_capacity, da_capacity = compute_archive_capacities(pop_size)
    new_state = archive_evict_ca(new_state, target_size=ca_capacity)
    new_state = archive_evict_da(new_state, target_size=da_capacity)

    return new_state, AdmissionReport(
        admitted_genome_ids=tuple(m.genome_id for m in admitted_filtered),
        n_mission=n_mission,
        n_progress=n_progress,
        n_bypass=len(bypass_admits),
        mode=mode,
        hard_constraint_drops=len(admitted) - len(admitted_filtered),
    )
```

### 8.3 ハード制約

```python
def apply_hard_constraints(
    admitted: Sequence[ArchiveMember],
    archive_state: ArchiveState,
    pop_size: int,
) -> list[ArchiveMember]:
    """per_run_max / pattern_max_share=0.25 / recency_floor=12 (last 3 runs) 適用.

    - per_run_max: 当 Run の admitted を per_run_max 体までトリム
      (mission_pass 優先、 次 progress_pass、 次 score_bypass の階層順)
    - pattern_max_share=0.25: 同 pattern_id の admitted は archive_total × 0.25 まで
    - recency_floor=12: 直近 3 Run 由来は最低 12 体保持 (eviction 側で対応、 admission では制約なし)
    """
    targets = compute_inflow_targets(pop_size, ...)
    # 1. per_run_max トリム
    if len(admitted) > targets.per_run_max:
        admitted = sorted(admitted, key=_archive_role_priority)[:targets.per_run_max]
    # 2. pattern_max_share
    archive_total = compute_archive_capacities(pop_size)[0]
    pattern_cap = int(archive_total * 0.25)
    pattern_count = Counter(m.pattern_id for m in archive_state.members)
    filtered = []
    for m in admitted:
        if pattern_count[m.pattern_id] < pattern_cap:
            filtered.append(m)
            pattern_count[m.pattern_id] += 1
    return filtered
```

注: pattern_max_share / recency_floor の precise 仕様は詳細設計で確定。 概念設計では「ハード制約の存在」 と admission 順序の優先付けのみ確定。

### 8.4 score_bypass の品質床 (margin_inf percentile <= 70)

`MARGIN_INF_PERCENTILE_70_THRESHOLD` は当 Run の全 candidates から動的に計算 (caller 責務、 T067 や T071 で実装):

```
margin_inf 値の昇順 sort で 70 percentile 値を threshold 化
margin_inf <= threshold の個体のみ score_bypass 候補
```

T066 では引数で `quality_floor_margin: Mapping[int, float]` を受け取り、 各候補が threshold を通過したかを `m.margin_inf <= threshold` で判定する。 threshold 計算自体は別 helper。

### 9.1 CA eviction (lex 8 段、 synthesis Round 21 改訂後)

```python
def _ca_eviction_sort_key(m: ArchiveMember) -> tuple:
    """CA lex eviction key (大が上位 = 残る、 末尾から削除).

    Round 21 改訂後の SSOT:
    1. mission_pass (True が上位)
    2. progress_pass (True が上位)
    3. not_score_bypass (True が上位)
    4. c_pass_depth (大が上位)
    5. mission_signed_margin (大が上位、 T062)
    6. shadow_robustness_score (大が上位、 T064)
    7. recency (新が上位 = run_id_index 降順)
    8. log_pf_clip (大が上位、 T061)
    """
    return (
        not m.mission_pass,                # False (= 0) が前 = 上位 = 残る
        not m.progress_pass,
        not m.not_score_bypass,
        -m.c_pass_depth,
        -m.mission_signed_margin,
        -m.shadow_robustness_score,
        -m.run_id_index,                   # recency: 新 = 大 = 残る
        -m.log_pf_clip,
    )


def archive_evict_ca(state: ArchiveState, *, target_size: int) -> ArchiveState:
    """CA capacity 超過時、 lex 末尾から削除して target_size 体に."""
    ca_members = [m for m in state.members if m.archive_target == "CA"]
    if len(ca_members) <= target_size:
        return state
    sorted_ca = sorted(ca_members, key=_ca_eviction_sort_key)   # 上位ほど残る
    survivors = sorted_ca[:target_size]
    survivor_ids = {m.genome_id for m in survivors}
    new_members = tuple(
        m for m in state.members
        if m.archive_target != "CA" or m.genome_id in survivor_ids
    )
    return replace(state, members=new_members)
```
