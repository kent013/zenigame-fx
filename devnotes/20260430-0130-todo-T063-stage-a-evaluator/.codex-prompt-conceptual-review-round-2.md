#### 2. unsigned divergence_offset_steps (Round 1 [C2])

```python
@dataclass(frozen=True)
class StageAControllerState:
    divergence_offset_steps: int   # unsigned >= 0
    last_a_b_correlation: float | None


def compute_q_force_with_divergence(base, steps: int) -> float:
    if steps < 0:
        raise ValueError(...)
    return min(base + 0.02 * steps, 0.40)


def update_divergence_state(prev, corr: float) -> StageAControllerState:
    if corr < 0.5:
        new_steps = clamp(prev.divergence_offset_steps + 1, 0, 13)
    else:
        new_steps = max(0, prev.divergence_offset_steps - 1)
    return StageAControllerState(divergence_offset_steps=new_steps, last_a_b_correlation=corr)
```
