    # legacy fold 計算 try 内 (bt 取得後):

        if bt.total_pnl is not None:
            _candidate_pnl = float(bt.total_pnl)
            if math.isfinite(_candidate_pnl):

    # try ブロック外、 既存 fold_sharpe is None / else 分岐:
