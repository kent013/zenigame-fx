## 修正後の核心箇所

### §2.3 probe_one (verdict 仕様表 + 二重防御)

```python
def probe_one(client: OandaClient, instrument: str) -> ProbeResult:
    try:
        resp = client.get_candles(instrument, granularity="M1", price="BA", count=10)
    except OandaAuthError:
        raise
    except httpx.HTTPStatusError as exc:
        status_code = exc.response.status_code
        if status_code == 401:
            raise OandaAuthError(f"401 Unauthorized: {exc.response.text[:200]}") from exc
        verdict = "FORBIDDEN" if status_code==403 else "NOT_FOUND" if status_code==404 else "OTHER"
        return ProbeResult(instrument, status_code, verdict, None, None, exc.response.text[:500])
    except Exception as exc:
        return ProbeResult(instrument, None, "OTHER", None, None, f"{type(exc).__name__}: {exc}"[:500])
    first_time = resp.candles[0].time.isoformat() if resp.candles else None
    return ProbeResult(instrument, 200, "OK", len(resp.candles), first_time, None)
```

### §2.4 run_probe (settings 再評価 + account_id/token 注入)

```python
def _resolve_runtime_settings(base_url, account_id, token):
    from src.config import Settings  # 遅延 import + 都度 instantiate
    s = Settings()
    return (base_url or s.oanda_base_url, account_id or s.oanda_account_id, token or s.oanda_api_token)

def run_probe(instruments, output_json, output_md, base_url=None, account_id=None, token=None) -> int:
    probed_at = datetime.now(tz=UTC)
    results = []
    actual_base_url, actual_account_id, actual_token = _resolve_runtime_settings(base_url, account_id, token)
    try:
        with OandaClient(base_url=actual_base_url, account_id=actual_account_id, token=actual_token) as client:
            for name in instruments:
                results.append(probe_one(client, name))
                _print_stdout(results[-1])
    except OandaAuthError as exc:
        print(f"[fatal] OANDA 401 Unauthorized: {exc}", file=sys.stderr)
        return 1
    summary = ProbeSummary.from_results(results)
    _write_json(output_json, probed_at, actual_base_url, results, summary)
    _write_markdown(output_md, probed_at, actual_base_url, results, summary)
    return 0
```

### §2.4 _write_json (probed_at 正規化)

```python
def _write_json(path, probed_at, base_url, results, summary):
    payload = {
        "probed_at": probed_at.isoformat(),  # str に正規化
        "base_url": base_url,
        "results": [asdict(r) for r in results],  # ProbeResult field は str/int/None のみ
        "summary": asdict(summary),
    }
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
```

### §4.1 runbook Verdict 仕様表

| HTTP status | verdict | スクリプト挙動 |
|---|---|---|
| 200 | OK | 結果記録、続行 |
| 401 | (fatal) | 即 abort, exit 1, JSON/MD は書かない |
| 403 | FORBIDDEN | 結果記録、続行 |
| 404 | NOT_FOUND | 結果記録、続行 |
| 5xx (retry 失敗後) / 429 / TransportError / その他 | OTHER | 結果記録、続行 |
