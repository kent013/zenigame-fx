### 施策 3: selection_score 6 要素化 + finite guard
```python
@property
def selection_score(self) -> tuple[int, float, int, int, int, float]:
    v = self.violation_magnitude
    v_norm = math.inf if not math.isfinite(v) else float(v)
    fp = self.fitness_pen
    fp_norm = -math.inf if not math.isfinite(fp) else float(fp)
    return (
        int(self.feasible),
        -v_norm,
        int(self.stage_c_pass),
        int(self.stage_b_pass),
        int(self.stage_a_pass),
        fp_norm,
    )
```

### 施策 4: 共通 fallback 関数
```python
def _selection_key(entry, fallback_active):
    if fallback_active:
        fp = entry.fitness_pen if math.isfinite(entry.fitness_pen) else -math.inf
        return (int(entry.stage_c_pass), int(entry.stage_b_pass), int(entry.stage_a_pass), fp)
    return entry.selection_score

def _is_all_infeasible(entries, feasibility_cfg):
    if not feasibility_cfg.enable_fallback_when_all_infeasible: return False
    return not any(e.feasible for e in entries)
```
`_tournament`, `_breed_next_gen` (sorted), `_select_best` で `fallback = _is_all_infeasible(cache.values(), cfg)` を毎世代計算し共通使用。
