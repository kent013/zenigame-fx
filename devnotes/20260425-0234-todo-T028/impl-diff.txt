+def _make_oscillating_bars(n_days: int, bars_per_day: int = 4) -> list[PriceBar]:
+    """value が 1 bar ごとに up/down する bars (取引によって PnL が出やすい)。"""
+    bars: list[PriceBar] = []

+            offset = (n % 4) - 2

+                    ask=Ohlc(ask_close, ask_close, ask_close, ask_close),
+                    volume=10,
+                    complete=True,

+def test_position_is_frozen() -> None:
+    """Position が @dataclass(frozen=True) で field 書き換えが禁止されていること。"""
+    pos = Position(

+        pos.entry_price = Decimal("155.00")  # type: ignore[misc]
