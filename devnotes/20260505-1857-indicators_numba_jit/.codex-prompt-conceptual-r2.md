```python
@numba.njit(cache=True, fastmath=False)
def _wilder_smooth(values: np.ndarray, n: int) -> np.ndarray:
    length = values.shape[0]
    out = np.full(length, np.nan, dtype=np.float64)
    if length < n:
        return out
    has_nan = False
    s = 0.0
    cnt = 0
    for k in range(n):
        v = values[k]
        if v == v:  # not NaN
            s += v
            cnt += 1
        else:
            has_nan = True
    if cnt == 0:
        return out
    seed = s / cnt
    out[n - 1] = seed
    prev = seed
    inv_n = 1.0 / n
    for i in range(n, length):
        v = values[i]
        if v == v:
            cur = inv_n * v + (1.0 - inv_n) * prev
            out[i] = cur
            prev = cur
        else:
            out[i] = prev  # 途中 NaN は前値維持
    return out
```
