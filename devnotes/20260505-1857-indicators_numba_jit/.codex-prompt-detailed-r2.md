### 施策 3 (`_wilder_smooth`) 修正: seed Python 維持 + recurrence のみ JIT 化

```python
def _wilder_smooth(values: np.ndarray, n: int) -> np.ndarray:
    if n <= 0:
        raise ValueError(f"n must be >= 1, got {n}")
    values = np.asarray(values, dtype=np.float64)
    length = len(values)
    out = np.full(length, np.nan, dtype=np.float64)
    if length < n:
        return out
    # seed: 現行と完全に同一の演算順序 (exact parity 保証)
    seed = float(np.nanmean(values[:n])) if np.any(np.isnan(values[:n])) else float(values[:n].mean())
    if not np.isfinite(seed):
        return out  # 全 NaN 等で seed=NaN → out 全 NaN のまま return
    out[n - 1] = seed
    # recurrence loop のみ Numba JIT 化
    _wilder_smooth_recurrence_jit(values, out, n, seed)
    return out


@numba.njit(cache=True, fastmath=False)
def _wilder_smooth_recurrence_jit(
    values: np.ndarray,
    out: np.ndarray,
    n: int,
    seed: float,
) -> None:
    """In-place: out[n:length] を recurrence で埋める。"""
    length = values.shape[0]
    prev = seed
    inv_n_m1 = float(n - 1)
    inv_n = float(n)
    for i in range(n, length):
        v = values[i]
        if v == v:
            cur = (prev * inv_n_m1 + v) / inv_n
            out[i] = cur
            prev = cur
        else:
            out[i] = prev
```
