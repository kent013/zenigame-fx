### 施策 1: rolling_max Numba JIT (circular buffer)
```python
@numba.njit(cache=True, fastmath=False)
def _rolling_max_jit(values: np.ndarray, n: int) -> np.ndarray:
    length = values.shape[0]
    out = np.full(length, np.nan, dtype=np.float64)
    dq = np.empty(n, dtype=np.int64)
    head = 0
    tail = 0
    size = 0
    for i in range(length):
        # window 外を front から drop
        while size > 0 and dq[head] <= i - n:
            head = (head + 1) % n
            size -= 1
        # 小さい値を back から drop (`<=` で等値も drop = 現行と一致)
        while size > 0:
            back_idx = (tail - 1 + n) % n
            if values[dq[back_idx]] <= values[i]:
                tail = back_idx
                size -= 1
            else:
                break
        dq[tail] = i
        tail = (tail + 1) % n
        size += 1
        if i >= n - 1:
            out[i] = values[dq[head]]
    return out
```

### 施策 3: _wilder_smooth Numba JIT
```python
@numba.njit(cache=True, fastmath=False)
def _wilder_smooth_jit(values: np.ndarray, n: int) -> np.ndarray:
    length = values.shape[0]
    out = np.full(length, np.nan, dtype=np.float64)
    if length < n:
        return out
    # seed: np.nanmean 相当を手動実装
    s = 0.0
    cnt = 0
    for k in range(n):
        v = values[k]
        if v == v:  # not NaN
            s += v
            cnt += 1
    if cnt == 0:
        return out  # 全 NaN
    seed = s / cnt
    out[n - 1] = seed
    prev = seed
    inv_n_m1 = float(n - 1)
    inv_n = float(n)
    for i in range(n, length):
        v = values[i]
        if v == v:
            cur = (prev * inv_n_m1 + v) / inv_n
            out[i] = cur
            prev = cur
        else:
            out[i] = prev  # 途中 NaN は前値維持
    return out
```
