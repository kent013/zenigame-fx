@dataclass(frozen=True)
class LaneEvalContext:
    lane_id: str
    bars_a: tuple[PriceBar, ...]
    bars_b: tuple[PriceBar, ...]
    bars_holdout: tuple[PriceBar, ...]
    meta: InstrumentMeta
    bt_cfg: BacktestConfig
    cp_inputs: dict[str, Any] | None = None
    preflight_underfilled: bool = False
    preflight_payload: Mapping[str, Any] | None = None  # n_unique_dates / wf_min_dates / max_folds 等

    def __post_init__(self):
        # 浅い不変化を契約として強制 (Round 1 §1)
        assert isinstance(self.bars_a, tuple)
        assert isinstance(self.bars_b, tuple)
        assert isinstance(self.bars_holdout, tuple)
        if self.preflight_underfilled and self.preflight_payload is None:
            raise ValueError("preflight_payload required when preflight_underfilled=True")

### LaneManager._collect_results_in_population_order (世代番号固定)

```python
def _collect_results_in_population_order(self, lane, results, ...):
    generation_index = lane.generation_count  # ループ前ローカル固定
    stage_a_pass = stage_b_pass = stage_c_pass = graduation_count = 0
    stage_a_seconds_list = []
    # ...
    for genome, r in zip(lane.population, results):
        # error 経路: fail-closed StageResult 変換
        if r.error is not None:
            a_result_for_collect = StageResult(stage="A", passed=False,
                                                metrics={"genome_name": genome.name},
                                                reason_codes=("worker_error",))
        else:
            a_result_for_collect = r.stage_a
        self._archive.collect_stage_a(genome, lane.lane_id, generation_index,
                                       a_result_for_collect, ...)
        if self._diagnostics:
            self._diagnostics.record_stage_a(lane.lane_id, generation_index,
                                              genome.name, a_result_for_collect)
        if a_result_for_collect.metrics.get("wall_time_seconds"):
            stage_a_seconds_list.append(float(a_result_for_collect.metrics["wall_time_seconds"]))
        if not a_result_for_collect.passed:
            continue
        stage_a_pass += 1

        # preflight_underfilled なら main 側で偽 Stage B 結果生成
        if r.stage_b is None and ctx.preflight_underfilled:
            b_result = self._build_preflight_b_result(genome, ctx)
        elif r.stage_b is not None:
            b_result = r.stage_b
        else:
            continue  # worker error after Stage A pass (rare)
        self._archive.collect_stage_b(genome, lane.lane_id, generation_index, b_result)
        # ... record_stage_b / Stage C / graduation_criteria / _mark_for_graduation も generation_index 使用
    lane.generation_count += 1
    return {
        "lane_id": lane.lane_id, "state": lane.state, "n_evaluated": len(lane.population),
        "stage_a_pass": stage_a_pass, "stage_b_pass": stage_b_pass, "stage_c_pass": stage_c_pass,
        "graduation_count": graduation_count,
        "stage_a_seconds_total": sum(stage_a_seconds_list),
        "stage_a_seconds_max": max(stage_a_seconds_list, default=0.0),
        # ... stage_b / stage_c も同様
    }
```

### 反復一致テスト (Round 1 §1 Critical 対応)

@pytest.mark.parametrize("max_workers", [1, 2, 6])
def test_evaluation_is_deterministic_across_repetitions(max_workers, fixture_population):
    eval1 = GenomeEvaluator(max_workers, ...)
    eval2 = GenomeEvaluator(max_workers, ...)
    r1 = eval1.evaluate_population("tier1_X", 0, fixture_population)
    r2 = eval2.evaluate_population("tier1_X", 0, fixture_population)
    eval1.close(); eval2.close()
    for a, b in zip(r1, r2):
        assert a.stage_a.metrics["payload"]["fitness_pen"] == b.stage_a.metrics["payload"]["fitness_pen"]


@pytest.mark.parametrize("max_workers", [2, 6])
def test_parallel_matches_sequential_for_same_seed(max_workers, fixture_population):
    eval_seq = GenomeEvaluator(1, ...)
    eval_par = GenomeEvaluator(max_workers, ...)
    r_seq = eval_seq.evaluate_population("tier1_X", 0, fixture_population)
    r_par = eval_par.evaluate_population("tier1_X", 0, fixture_population)
    eval_seq.close(); eval_par.close()
    for s, p in zip(r_seq, r_par):
        assert s.genome_name == p.genome_name
        assert s.stage_a.passed == p.stage_a.passed
        for key in ("fitness", "fitness_pen", "trade_count"):
            assert s.stage_a.metrics["payload"].get(key) == p.stage_a.metrics["payload"].get(key)

### メモリ運用ガード測定

```python
def measure_peak_rss_mb() -> dict[str, float]:
    process = psutil.Process()
    main_rss_mb = process.memory_info().rss / 1024 / 1024
    children_rss_mb: list[float] = []
    for child in process.children(recursive=True):
        try:
            children_rss_mb.append(child.memory_info().rss / 1024 / 1024)
        except psutil.NoSuchProcess:
            continue
    return {
        "main_rss_mb": main_rss_mb,
        "max_rss_mb_per_worker": max(children_rss_mb, default=0.0),
        "total_rss_mb": main_rss_mb + sum(children_rss_mb),
        "n_children": len(children_rss_mb),
    }
```
