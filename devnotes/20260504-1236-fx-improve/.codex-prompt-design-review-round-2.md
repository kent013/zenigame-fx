## 改訂提案 (= 詳細設計 § C1 への delta)

### Delta 1: 共通 valid_mask 関数追加

```python
def compute_valid_mask(df: pd.DataFrame) -> pd.Series:
    """sentinel 除外 + finite + 必須列存在の共通 mask.
    全集計関数で再利用 (= R1-W2 対応)."""
    sentinel_threshold = -1e6  # NO_EXPOSURE / METRIC_UNAVAILABLE / SYSTEM_FAILURE 全て -1e9
    return (
        (df["fitness_pen"] > sentinel_threshold)
        & df["fitness_pen"].apply(lambda x: math.isfinite(x) if x is not None else False)
        & df["trade_sharpe_raw"].notna()
    )
```

### Delta 2: P2 閾値を比率併用に変更

```python
def diagnose_root_cause(
    decomposition: dict[str, Any],
    diversity: list[dict[str, Any]],
    penalty_effect: dict[str, Any],
    *,
    n_valid: int,
    max_clause_config: int,  # ← 追加
) -> tuple[str, list[str], dict[str, Any]]:
    """P1/P2/P3/INCONCLUSIVE + non_evaluable_reason."""
    rationale: list[str] = []
    falsification: dict[str, Any] = {}
    raw_max = decomposition.get("trade_sharpe_raw", {}).get("max")

    # P1: primitive 不足 (raw<0.005)
    p1_match = raw_max is not None and raw_max < 0.005
    falsification["P1"] = {"condition": "raw_max < 0.005", "actual": raw_max, "match": p1_match}

    # P2: penalty/config (raw>0 ∧ killed>=max(3, ceil(0.01*N_valid)))
    p2_threshold = max(3, math.ceil(0.01 * n_valid))
    p2_killed = penalty_effect["n_raw_positive_and_penalty_killed"]
    p2_match = p2_killed >= p2_threshold
    falsification["P2"] = {
        "condition": f"penalty_killed >= max(3, 0.01*N_valid={n_valid}) = {p2_threshold}",
        "actual": p2_killed, "match": p2_match,
    }

    # P3: 探索 dynamics (= max_clause<=1 で非識別 → N/A 化、 R1-Cr1 対応)
    p3_non_evaluable = max_clause_config <= 1
    if p3_non_evaluable:
        falsification["P3"] = {
            "condition": "n_nodes std drop > 0.5 ∧ active_clause unique == 1",
            "actual": "non_evaluable",
            "match": False,
            "non_evaluable_reason": (
                f"config.max_clause={max_clause_config} <= 1 のため "
                "active_clause unique==1 は config 由来の擬陽性。 P3 識別力なし"
            ),
        }
        p3_match = False
    elif diversity:
        first_std = diversity[0]["n_nodes_std"]
        last_std = diversity[-1]["n_nodes_std"]
        active_unique = diversity[-1]["active_clause_unique"]
        p3_match = (first_std - last_std) > 0.5 and active_unique == 1
        falsification["P3"] = {
            "condition": f"n_nodes std drop > 0.5 ∧ active_clause unique == 1",
            "actual": {"std_drop": first_std - last_std, "active_unique": active_unique},
            "match": p3_match,
        }
    else:
        p3_match = False

    # 集約 + 信頼度
    matches = [m for m, ok in [("P1", p1_match), ("P2", p2_match), ("P3", p3_match)] if ok]
    if len(matches) == 1:
        diagnosis = matches[0]
        confidence = "high" if not p3_non_evaluable or diagnosis != "P3" else "medium"
    elif len(matches) > 1:
        diagnosis = "INCONCLUSIVE"
        confidence = "low"
        rationale.append(f"複数仮説該当: {matches}")
    else:
        diagnosis = "INCONCLUSIVE"
        confidence = "low"
        rationale.append("どの仮説にも明確該当せず")

    return diagnosis, rationale, {
        "confidence": confidence,
        "falsification": falsification,
        "p3_non_evaluable": p3_non_evaluable,
    }
```

### Delta 3: plateau_length + sentinel breakdown 追加

```python
def compute_plateau_length(df: pd.DataFrame) -> int:
    """best fitness_pen が連続して同値だった世代数."""
    valid = df[compute_valid_mask(df)].copy()
    if len(valid) == 0:
        return 0
    by_gen = valid.groupby("generation")["fitness_pen"].max().sort_index()
    if len(by_gen) < 2:
        return 0
    last_value = by_gen.iloc[-1]
    plateau = 0
    for v in reversed(by_gen.tolist()):
        if v == last_value:
            plateau += 1
        else:
            break
    return plateau


def compute_sentinel_breakdown(df: pd.DataFrame) -> dict[str, int]:
    """sentinel 値別個体数 (= データ品質セクション)."""
    NO_EXPOSURE = -1e9  # NO_EXPOSURE_FITNESS (T034)
    METRIC_UNAVAILABLE = -1e9  # METRIC_UNAVAILABLE_FITNESS (T034)
    SYSTEM_FAILURE = -1e9  # SYSTEM_FAILURE_FITNESS (T034)
    # 全て -1e9 で区別不能のため total のみ集計 (= reason_codes が archive にないと分離不能)
    total_sentinel = int((df["fitness_pen"] <= -1e6).sum())
    return {
        "total_sentinel_individuals": total_sentinel,
        "note": (
            "T034 sentinel 値は全て -1e9 で同一、 archive に Stage A reason_codes 列なしのため "
            "NO_EXPOSURE / METRIC_UNAVAILABLE / SYSTEM_FAILURE の内訳分離不能"
        ),
    }
```
