def _compute(bars):
    n = len(bars)
    o = np.empty(n, dtype=np.float64)
    h = np.empty(n, dtype=np.float64)
    low = np.empty(n, dtype=np.float64)
    c = np.empty(n, dtype=np.float64)
    for i, b in enumerate(bars):  # enumerate 維持
        bid = b.bid  # 1 access (was 4)
        ask = b.ask  # 1 access (was 4)
        o[i] = (float(bid.open) + float(ask.open)) * 0.5
        h[i] = (float(bid.high) + float(ask.high)) * 0.5
        low[i] = (float(bid.low) + float(ask.low)) * 0.5
        c[i] = (float(bid.close) + float(ask.close)) * 0.5
    return o, h, low, c
